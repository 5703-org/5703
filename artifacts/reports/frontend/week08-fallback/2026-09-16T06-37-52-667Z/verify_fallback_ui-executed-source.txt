import fs from 'node:fs/promises';
import { createHash } from 'node:crypto';

  const result = await page.request.get(`/api/v1${path}`, { headers: { Authorization: `Bearer ${token}` } });
  if (!result.ok()) throw new Error(`Read-only API status ${result.status()}`);
  return (await result.json()).data;
}

  expect(before.counts.filtered).toBeGreaterThan(0);
  expect(before.counts.candidates).toBeLessThanOrEqual(20);
